Simple marginalia
================================================================================

/* application=super-app */
SELECT id /* MAX_EXECUTION_TIME(500) */
FROM my_table;

--------------------------------------------------------------------------------

(program
  (marginalia)
  (statement
    (select
      (keyword_select)
      (select_expression
        (term
          value: (field
            name: (identifier)))))
    (marginalia)
    (from
      (keyword_from)
      (relation
        (object_reference
          name: (identifier))))))
